const OCR_PROMPT = 'Analiza esta imagen para usarla como contexto de conversación. Describe con precisión el contenido visual y transcribe todo el texto legible (OCR), también la letra a mano, respetando títulos, listas y el orden. const TOKEN_SHA256 = '<sha256 de la llave de un solo uso>';
const MODEL = 'qwen/qwen3.8-27b';
const KEY_NAMES = ['GROQ_API_KEY_1', 'GROQ_API_KEY_2', 'GROQ_API_KEY_3', 'GROQ_API_KEY_4', 'GROQ_API_KEY'];

async function sha256(text: string) {
  const digest = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(text));
  return [...new Uint8Array(digest)].map((b) => b.toString(16).padStart(2, '0')).join('');
}
const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { 'Content-Type': 'application/json' } });
const keys = (names: string[]) => [...new Set(names.map((n) => Deno.env.get(n)?.trim()).filter((k): k is string => Boolean(k) && !/[\s"']/.test(k!)))];

async function groq(payload: Record<string, unknown>) {
  const started = Date.now();
  let last = 0;
  for (const key of keys(KEY_NAMES)) {
    const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST', signal: AbortSignal.timeout(90000),
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${key}` },
      body: JSON.stringify(payload),
    });
    last = res.status;
    if (res.status === 429 || res.status >= 500 || res.status === 401) { await res.body?.cancel(); continue; }
    const data = await res.json();
    if (!res.ok) return json({ error: 'groq_error', status: res.status, message: String(data?.error?.message ?? '').slice(0, 300) }, 502);
    return json({ content: data.choices?.[0]?.message?.content ?? '', usage: data.usage ?? null, ms: Date.now() - started,
      remaining: { tokens_minute: res.headers.get('x-ratelimit-remaining-tokens') } });
  }
  return json({ error: 'groq_unavailable', status: last }, 503);
}

Deno.serve(async (req) => {
  if (req.method !== 'POST') return json({ error: 'not_found' }, 404);
  if ((await sha256(req.headers.get('x-banco-token') ?? '')) !== TOKEN_SHA256) return json({ error: 'not_found' }, 404);
  let body: any;
  try { body = await req.json(); } catch { return json({ error: 'bad_json' }, 400); }
  if (body.action === 'groq') {
    const messages = Array.isArray(body.messages) ? body.messages.slice(0, 6).map((m: any) => ({ role: ['system', 'user', 'assistant'].includes(m?.role) ? m.role : 'user', content: String(m?.content ?? '').slice(0, 20000) })) : [];
    if (!messages.length) return json({ error: 'bad_messages' }, 400);
    return groq({ model: MODEL, messages, temperature: 0.2, stream: false, max_completion_tokens: 2400, reasoning_format: 'hidden', reasoning_effort: 'none',
      response_format: { type: 'json_schema', json_schema: { name: 'apuntes', strict: false, schema: NOTES_JSON_SCHEMA } } });
  }
  if (body.action === 'vision') {
    const image = String(body.image ?? '');
    if (!/^[A-Za-z0-9+/=]{100,1500000}$/.test(image)) return json({ error: 'bad_image' }, 400);
    return groq({ model: Deno.env.get('GROQ_VISION_MODEL')?.trim() || MODEL, stream: false, temperature: 0.2, max_completion_tokens: 1400,
      messages: [{ role: 'user', content: [{ type: 'text', text: OCR_PROMPT }, { type: 'image_url', image_url: { url: `data:image/jpeg;base64,${image}` } }] }] });
  }
  return json({ error: 'not_found' }, 404);
});
